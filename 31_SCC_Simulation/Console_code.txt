#!/bin/bash
# -------------------------------------------------------------
# Console_code.txt
# Part of the Model Predictive Control in buildings repository
# Developed by Roberto Garay Martinez
# -------------------------------------------------------------
# Location: 31_SCC_Simulation/Console_code.txt
# -------------------------------------------------------------
# Commands to execute on the SCC cluster to submit parametric
# MPC simulations as Slurm job arrays.
# -------------------------------------------------------------
# Usage:
#   bash "31_SCC_Simulation/Console_code.txt"        # all jobs
#   bash "31_SCC_Simulation/Console_code.txt" 2      # only 2nd
#   bash "31_SCC_Simulation/Console_code.txt" 0      # all jobs
#
# The optional positional argument sets the job selector n.
# n=0 (default) submits all jobs; n=K submits only the K-th.
# -------------------------------------------------------------
# The TASKS_PER_JOB variable sets how many tasks (individual
# simulation configurations) are grouped into a single Slurm
# job array submission.
# JOB_NAME sets the Slurm --job-name for easy identification.
# Both are configurable from the parametric simulation GUI
# and stored in scc_settings.sh.
# If scc_settings.sh does not exist, default values are used.
# -------------------------------------------------------------
# SCC_QUEUE, SCC_QOS, SCC_USER and SCC_USERNAME are the Slurm
# partition, Quality of Service, user email for notifications,
# and cluster login username (used for squeue monitoring). All
# are configurable from the GUI and stored in scc_settings.sh.
# -------------------------------------------------------------
# SCC_KEEP_LOGS controls whether Slurm .err/.out log files are
# kept (1) or deleted (0) after each job array finishes.
# SCC_MAX_REQUEUE sets how many times failed configurations are
# automatically re-submitted after all jobs in the array finish.
# Both are configurable from the GUI and stored in scc_settings.sh.
# -------------------------------------------------------------
# File management (DIPC SCC):
#   The project directory lives on /dipc/USERNAME/JOB_NAME.
#   Before simulation, it is synced to /scratch/USERNAME/JOB_NAME
#   (the fast local storage of the compute node) using rsync and
#   execution happens from there. After job submission the user
#   can monitor with squeue.
# -------------------------------------------------------------
# Job persistence:
#   A job tracker (job_tracker.csv) records every job's status.
#   Each finished job auto-launches the next pending job, so
#   all jobs complete without manual intervention even when
#   the queue limit is reached.
# -------------------------------------------------------------
# Note: CRLF to LF conversion is performed at the start of
# this script to prevent sbatch errors when files were
# generated on Windows or transferred with DOS line endings.
# -------------------------------------------------------------

# --- Job selector (positional argument overrides default) ------
# Set n=0 (or leave empty) to submit ALL jobs.
# Set n=1 for the 1st job, n=2 for the 2nd job, etc.
# Pass as argument: bash "31_SCC_Simulation/Console_code.txt" 2
n=${1:-0}

# --- Convert DOS (CRLF) to Unix (LF) line endings ---------------
# Files may have been generated on Windows or transferred with DOS
# line endings. This conversion ensures compatibility with Linux
# and prevents sbatch errors like "Batch script contains DOS line
# breaks". This is safe to run even if the files already have LF.
# -----------------------------------------------------------------
sed -i 's/\r$//' "31_SCC_Simulation/Console_code.txt"
sed -i 's/\r$//' "31_SCC_Simulation/Job_array_r.sh"
[[ -f "31_SCC_Simulation/scc_settings.sh" ]]      && sed -i 's/\r$//' "31_SCC_Simulation/scc_settings.sh"
[[ -f "31_SCC_Simulation/Optim_parameters.csv" ]]  && sed -i 's/\r$//' "31_SCC_Simulation/Optim_parameters.csv"

# --- Load SCC settings (generated by parametric simulation GUI) ---
SCC_SETTINGS="31_SCC_Simulation/scc_settings.sh"
if [[ -f "$SCC_SETTINGS" ]]; then
  source "$SCC_SETTINGS"
else
  TASKS_PER_JOB=100
  JOB_NAME="MPC_optim"
  SCC_QUEUE="general"
  SCC_QOS="serial"
  SCC_USER=""
  SCC_USERNAME=""
  SCC_OUTPUT_FORMATS="csv,rds"
  echo "Warning: $SCC_SETTINGS not found. Using defaults."
fi

# Apply defaults if variables are empty
: "${TASKS_PER_JOB:=100}"
: "${JOB_NAME:=MPC_optim}"
: "${SCC_QUEUE:=general}"
: "${SCC_QOS:=serial}"
: "${SCC_USER:=}"
: "${SCC_USERNAME:=}"
: "${SCC_OUTPUT_FORMATS:=csv,rds}"
: "${SCC_KEEP_LOGS:=1}"
: "${SCC_MAX_REQUEUE:=2}"

# --- Copy project to /scratch (fast node-local storage) --------
# On the DIPC SCC the user home is /dipc/USERNAME. The project
# directory is /dipc/USERNAME/JOB_NAME. We copy it to the fast
# scratch filesystem before execution and cd into it.
# ---------------------------------------------------------------
DIPC_USER="${SCC_USERNAME:-$USER}"
SRC_DIR="/dipc/${DIPC_USER}/${JOB_NAME}"
DST_DIR="/scratch/${DIPC_USER}/${JOB_NAME}"

echo "Syncing project to scratch: ${SRC_DIR} -> ${DST_DIR}"
mkdir -p "${DST_DIR}"
rsync -a --delete "${SRC_DIR}/" "${DST_DIR}/"
cd "${DST_DIR}" || { echo "Cannot cd to ${DST_DIR}" >&2; exit 1; }
echo "Working directory: $(pwd)"

# --- Create log directory ---
mkdir -p logs

# --- Count configurations ---
INPUT_FILE="31_SCC_Simulation/Optim_parameters.csv"
[[ -f "$INPUT_FILE" ]] || { echo "Not found: $INPUT_FILE" >&2; exit 1; }

N=$(( $(wc -l < "$INPUT_FILE") - 1 ))
echo "Number of configurations (tasks): $N"
echo "Tasks per job: $TASKS_PER_JOB"
echo "Job name: $JOB_NAME"
echo "Slurm partition (queue): $SCC_QUEUE"
echo "Slurm QoS: $SCC_QOS"
echo "User email (mail-user): ${SCC_USER:-(not set)}"
echo "Username (squeue): ${SCC_USERNAME:-(not set, using \$USER)}"
echo "Output formats: $SCC_OUTPUT_FORMATS"

TOTAL_JOBS=$(( (N + TASKS_PER_JOB - 1) / TASKS_PER_JOB ))
echo "Total jobs to submit: $TOTAL_JOBS"

if [[ "$n" -gt 0 ]]; then
  echo "Submitting ONLY job: $n"
else
  echo "Submitting ALL jobs"
fi

# --- Load R module ---
module spider R
module load R/4.2.1-foss-2022a

# --- Install libraries (only needed once) ---
Rscript "31_SCC_Simulation/Install_libraries.R"

# --- Build sbatch extra arguments for partition, qos, mail-user and job-name ---
SBATCH_EXTRA=""
if [[ -n "$JOB_NAME" ]]; then
  SBATCH_EXTRA="$SBATCH_EXTRA --job-name=$JOB_NAME"
fi
if [[ -n "$SCC_QUEUE" ]]; then
  SBATCH_EXTRA="$SBATCH_EXTRA --partition=$SCC_QUEUE"
fi
if [[ -n "$SCC_QOS" ]]; then
  SBATCH_EXTRA="$SBATCH_EXTRA --qos=$SCC_QOS"
fi
if [[ -n "$SCC_USER" ]]; then
  SBATCH_EXTRA="$SBATCH_EXTRA --mail-user=$SCC_USER"
fi

# Export output formats so Job_array_r.sh can pass it to R
export SCC_OUTPUT_FORMATS
export SCC_KEEP_LOGS
export SCC_MAX_REQUEUE

# --- Initialise job tracker ------------------------------------
# The tracker records which jobs have been launched, which have
# completed successfully, and which are still pending.
# Columns: job_index, start_task, end_task, status, slurm_job_id
# Statuses: pending, launched, completed
# ---------------------------------------------------------------
TRACKER="31_SCC_Simulation/job_tracker.csv"
echo "job_index,start_task,end_task,status,slurm_job_id" > "$TRACKER"

IDX=0
S=1
while [ $S -le $N ]; do
  E=$(( S + TASKS_PER_JOB - 1 ))
  if [ $E -gt $N ]; then
    E=$N
  fi
  IDX=$(( IDX + 1 ))
  echo "${IDX},${S},${E},pending," >> "$TRACKER"
  S=$(( E + 1 ))
done
echo "Job tracker initialised with $IDX entries: $TRACKER"

# --- Submit jobs (each with up to TASKS_PER_JOB tasks) ---
START=1
JOB_COUNT=0
JOB_INDEX=0
while [ $START -le $N ]; do
  END=$(( START + TASKS_PER_JOB - 1 ))
  if [ $END -gt $N ]; then
    END=$N
  fi
  JOB_INDEX=$(( JOB_INDEX + 1 ))

  # If n>0, only submit the nth job
  if [[ "$n" -gt 0 && "$JOB_INDEX" -ne "$n" ]]; then
    START=$(( END + 1 ))
    continue
  fi

  JOB_COUNT=$(( JOB_COUNT + 1 ))
  echo "Submitting job $JOB_INDEX: tasks ${START}-${END}${SBATCH_EXTRA:+ $SBATCH_EXTRA}"
  SBATCH_OUT=$(sbatch --array=${START}-${END} $SBATCH_EXTRA --export=ALL "31_SCC_Simulation/Job_array_r.sh" 2>&1)
  SBATCH_EXIT=$?
  echo "$SBATCH_OUT"

  # Extract Slurm job ID from sbatch output, then verify the job is
  # actually present in the queue.  Slurm may reject a submission
  # (quota, policy, invalid parameters, …) even after returning a
  # job ID, so we confirm with squeue before marking as launched.
  # If the check fails the entry stays "pending" and will be
  # retried automatically when the next running job finishes.
  # Use [0-9][0-9]* (at least one digit) to avoid matching empty strings.
  SLURM_JID=$(echo "$SBATCH_OUT" | grep -oE '[0-9]+$' || true)
  if [[ "$SBATCH_EXIT" -ne 0 ]]; then
    echo "Warning: sbatch rejected job $JOB_INDEX (exit $SBATCH_EXIT). Keeping as pending."
  elif [[ -n "$SLURM_JID" ]]; then
    # Slurm may take a brief moment to register the job in squeue after
    # sbatch returns; retry a few times before concluding it is absent.
    SLURM_QUEUED=0
    for _i in 1 2 3 4 5; do
      if squeue --noheader -j "$SLURM_JID" 2>/dev/null | grep -q .; then
        SLURM_QUEUED=1
        break
      fi
      sleep 2
    done
    if [[ "$SLURM_QUEUED" -eq 1 ]]; then
      sed -i "s/^${JOB_INDEX},${START},${END},pending,/${JOB_INDEX},${START},${END},launched,${SLURM_JID}/" "$TRACKER"
    else
      echo "Warning: job $JOB_INDEX (Slurm ID $SLURM_JID) not found in Slurm queue. Keeping as pending."
    fi
  else
    echo "Warning: sbatch output for job $JOB_INDEX contained no job ID. Keeping as pending."
  fi

  START=$(( END + 1 ))
done

echo "Total jobs submitted: $JOB_COUNT"

# --- Monitor queue ---
# Use SCC_USERNAME if set, otherwise fall back to $USER
SQUEUE_USER="${SCC_USERNAME:-$USER}"
squeue -u "$SQUEUE_USER"

# --- Generate initial operation summary in /dipc -------------------
# Write the initial summary (just after submission) to /dipc so
# the user can track progress before all jobs complete.
# ------------------------------------------------------------------
{
  DIPC_USER="${SCC_USERNAME:-$USER}"
  SUMMARY_DIR="/dipc/${DIPC_USER}/${JOB_NAME}/31_SCC_Simulation"
  mkdir -p "$SUMMARY_DIR"
  SUMMARY_FILE="${SUMMARY_DIR}/operation_summary_initial_$(date +%Y%m%d_%H%M%S).txt"

  {
    echo "==================================="
    echo "SCC Operation Summary (initial)"
    echo "==================================="
    echo "Date/Time (script start):      $(date)"
    echo "Job name:                      ${JOB_NAME}"
    echo "Working directory:             $(pwd)"
    echo "-----------------------------------"
    echo "Simulations queued at start:   ${N}"
    echo "Tasks per job:                 ${TASKS_PER_JOB}"
    echo "Total jobs submitted:          ${JOB_COUNT}"
    echo "Slurm partition:               ${SCC_QUEUE}"
    echo "Slurm QoS:                     ${SCC_QOS}"
    echo "Output formats:                ${SCC_OUTPUT_FORMATS}"
    echo "Keep logs:                     ${SCC_KEEP_LOGS:-1}"
    echo "Max re-queue attempts:         ${SCC_MAX_REQUEUE:-2}"
    echo "==================================="
  } > "$SUMMARY_FILE"

  echo "Initial summary written to: $SUMMARY_FILE"

  rsync -a "31_SCC_Simulation/" "/dipc/${DIPC_USER}/${JOB_NAME}/31_SCC_Simulation/"
}
