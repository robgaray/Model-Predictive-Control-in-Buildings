#!/bin/bash
# -------------------------------------------------------------
# Console script: Verification_code.txt
# Part of the Model Predictive Control in buildings repository
# https://github.com/robgaray/Model-Predictive-Control-in-Buildings_WORK4
# Developed by Roberto Garay Martinez
# -------------------------------------------------------------
# Location: 02_SCC_simulation/Verification_code.txt
# -------------------------------------------------------------
# Overall description
# This script verifies the correct execution of parametric MPC
# simulations on the SCC supercomputer. It is intended to be
# run manually by the user once all Slurm jobs have finished.
#
# It must be executed from the project root directory on
# /scratch/<USERNAME>/<JOB_NAME>.
# -------------------------------------------------------------
# Actions performed:
#   1. Load SCC settings
#   2. Backup Optim_parameters.csv
#   3. For each configuration in Optim_parameters.csv:
#      a. Check that the expected output file(s) exist
#      b. Validate file content: dataframe structure, column
#         names, column types, row count, and absence of NaN
#   4. Update Optim_parameters.csv to contain only the
#      configurations that failed verification
#   5. Print a summary report
# -------------------------------------------------------------
# Usage:
#   cd /scratch/<USERNAME>/<JOB_NAME>
#   bash "02_SCC_simulation/Verification_code.txt"
# -------------------------------------------------------------
# Inputs
#   02_SCC_simulation/Optim_parameters.csv
#   02_SCC_simulation/scc_settings.sh
#   01_Simulation/90_Output/Sinthetized_df_computed_*.csv/.rds
# -------------------------------------------------------------
# Outputs
#   02_SCC_simulation/Optim_parameters_backup_<timestamp>.csv
#   02_SCC_simulation/Optim_parameters.csv (updated: only failed)
#   Console report with verification results
# -------------------------------------------------------------

# --- Convert DOS (CRLF) to Unix (LF) line endings ---------------
sed -i 's/\r$//' "02_SCC_simulation/Verification_code.txt"
[[ -f "02_SCC_simulation/scc_settings.sh" ]]      && sed -i 's/\r$//' "02_SCC_simulation/scc_settings.sh"
[[ -f "02_SCC_simulation/Optim_parameters.csv" ]]  && sed -i 's/\r$//' "02_SCC_simulation/Optim_parameters.csv"

# --- 1. Load SCC settings ----------------------------------------
SCC_SETTINGS="02_SCC_simulation/scc_settings.sh"
if [[ -f "$SCC_SETTINGS" ]]; then
  source "$SCC_SETTINGS"
else
  SCC_OUTPUT_FORMATS="csv,rds"
  echo "Warning: $SCC_SETTINGS not found. Using default output formats."
fi
: "${SCC_OUTPUT_FORMATS:=csv,rds}"

INPUT_FILE="02_SCC_simulation/Optim_parameters.csv"
OUTPUT_DIR="01_Simulation/90_Output"

[[ -f "$INPUT_FILE" ]] || { echo "Not found: $INPUT_FILE" >&2; exit 1; }

# Load R module (needed for Rscript verification)
module spider R 2>/dev/null
module load R/4.2.1-foss-2022a 2>/dev/null

# Add local library path if present
LIB_PATH=""
if [[ -d "00_Libraries" ]]; then
  LIB_PATH="00_Libraries"
fi

# --- 2. Backup Optim_parameters.csv --------------------------------
TIMESTAMP=$(date +%Y%m%d_%H%M%S)
BACKUP_FILE="02_SCC_simulation/Optim_parameters_backup_${TIMESTAMP}.csv"
cp "$INPUT_FILE" "$BACKUP_FILE"
echo "Backup created: $BACKUP_FILE"

# --- 3. Count configurations ---------------------------------------
N=$(( $(wc -l < "$INPUT_FILE") - 1 ))
echo "==================================="
echo "Verification started at $(date)"
echo "Number of configurations to verify: $N"
echo "Output directory: $OUTPUT_DIR"
echo "Output formats: $SCC_OUTPUT_FORMATS"
echo "==================================="

# --- 4. Determine which output formats to check --------------------
CHECK_SYNTH_CSV=0
CHECK_SYNTH_RDS=0
CHECK_MAIN_CSV=0
CHECK_MAIN_RDS=0

IFS=',' read -ra FORMATS <<< "$SCC_OUTPUT_FORMATS"
for fmt in "${FORMATS[@]}"; do
  fmt=$(echo "$fmt" | xargs)
  case "$fmt" in
    synth_csv) CHECK_SYNTH_CSV=1 ;;
    synth_rds) CHECK_SYNTH_RDS=1 ;;
    main_csv)  CHECK_MAIN_CSV=1  ;;
    main_rds)  CHECK_MAIN_RDS=1  ;;
  esac
done

# --- 5. Read header from Optim_parameters.csv ----------------------
HEADER=$(head -n 1 "$INPUT_FILE")

# Prepare a temporary file for failed configurations
FAILED_FILE=$(mktemp)
echo "$HEADER" > "$FAILED_FILE"

PASS_COUNT=0
FAIL_COUNT=0

# --- 6. Verify each configuration ----------------------------------
# Read all data lines in a single pass (skip header) for efficiency.
# Process substitution (< <(...)) avoids a subshell so that
# PASS_COUNT, FAIL_COUNT, and FAILED_FILE updates are preserved.
CONT_001=0
while IFS= read -r LINE; do

  CONT_001=$(( CONT_001 + 1 ))

  # Parse CSV columns to build the filename suffix
  IFS=',' read -r popsize iter run pcross pmut opt_hor impl_hor anticip \
                  ctrl_type opt_aim flex_len flex_rec therm_stab min_flex \
                  flex_splits alpha month <<< "$LINE"

  # Trim whitespace
  popsize=$(echo "$popsize"         | xargs)
  iter=$(echo "$iter"               | xargs)
  run=$(echo "$run"                 | xargs)
  pcross=$(echo "$pcross"           | xargs)
  pmut=$(echo "$pmut"               | xargs)
  opt_hor=$(echo "$opt_hor"         | xargs)
  impl_hor=$(echo "$impl_hor"       | xargs)
  anticip=$(echo "$anticip"         | xargs)
  ctrl_type=$(echo "$ctrl_type"     | xargs)
  opt_aim=$(echo "$opt_aim"         | xargs)
  flex_len=$(echo "$flex_len"       | xargs)
  flex_rec=$(echo "$flex_rec"       | xargs)
  therm_stab=$(echo "$therm_stab"   | xargs)
  min_flex=$(echo "$min_flex"       | xargs)
  flex_splits=$(echo "$flex_splits" | xargs)
  alpha=$(echo "$alpha"             | xargs)
  month=$(echo "$month"             | xargs)

  SUFFIX="${popsize}_${iter}_${run}_${pcross}_${pmut}_${opt_hor}_${impl_hor}_${anticip}_${ctrl_type}_${opt_aim}_${flex_len}_${flex_rec}_${therm_stab}_${min_flex}_${flex_splits}_${alpha}_${month}"

  CONFIG_OK=1
  ERROR_MSGS=""

  # ------------------------------------------------------------------
  # 6.1 Check file existence
  # ------------------------------------------------------------------
  if [[ "$CHECK_SYNTH_CSV" -eq 1 ]]; then
    SYNTH_CSV="${OUTPUT_DIR}/Sinthetized_df_computed_${SUFFIX}.csv"
    if [[ ! -f "$SYNTH_CSV" ]]; then
      CONFIG_OK=0
      ERROR_MSGS="${ERROR_MSGS}  Missing: ${SYNTH_CSV}\n"
    fi
  fi

  if [[ "$CHECK_SYNTH_RDS" -eq 1 ]]; then
    SYNTH_RDS="${OUTPUT_DIR}/Sinthetized_df_computed_${SUFFIX}.rds"
    if [[ ! -f "$SYNTH_RDS" ]]; then
      CONFIG_OK=0
      ERROR_MSGS="${ERROR_MSGS}  Missing: ${SYNTH_RDS}\n"
    fi
  fi

  if [[ "$CHECK_MAIN_CSV" -eq 1 ]]; then
    MAIN_CSV="${OUTPUT_DIR}/Main_df_computed_${SUFFIX}.csv"
    if [[ ! -f "$MAIN_CSV" ]]; then
      CONFIG_OK=0
      ERROR_MSGS="${ERROR_MSGS}  Missing: ${MAIN_CSV}\n"
    fi
  fi

  if [[ "$CHECK_MAIN_RDS" -eq 1 ]]; then
    MAIN_RDS="${OUTPUT_DIR}/Main_df_computed_${SUFFIX}.rds"
    if [[ ! -f "$MAIN_RDS" ]]; then
      CONFIG_OK=0
      ERROR_MSGS="${ERROR_MSGS}  Missing: ${MAIN_RDS}\n"
    fi
  fi

  # ------------------------------------------------------------------
  # 6.2 Content verification (only if files exist)
  # Uses an R script to validate dataframe structure, types,
  # row count, and absence of NaN/NA values.
  # ------------------------------------------------------------------
  if [[ "$CONFIG_OK" -eq 1 ]]; then

    # Build the list of files to verify
    FILES_TO_CHECK=""
    if [[ "$CHECK_SYNTH_CSV" -eq 1 ]]; then
      FILES_TO_CHECK="${FILES_TO_CHECK},synth_csv:${OUTPUT_DIR}/Sinthetized_df_computed_${SUFFIX}.csv"
    fi
    if [[ "$CHECK_SYNTH_RDS" -eq 1 ]]; then
      FILES_TO_CHECK="${FILES_TO_CHECK},synth_rds:${OUTPUT_DIR}/Sinthetized_df_computed_${SUFFIX}.rds"
    fi
    if [[ "$CHECK_MAIN_CSV" -eq 1 ]]; then
      FILES_TO_CHECK="${FILES_TO_CHECK},main_csv:${OUTPUT_DIR}/Main_df_computed_${SUFFIX}.csv"
    fi
    if [[ "$CHECK_MAIN_RDS" -eq 1 ]]; then
      FILES_TO_CHECK="${FILES_TO_CHECK},main_rds:${OUTPUT_DIR}/Main_df_computed_${SUFFIX}.rds"
    fi
    # Remove leading comma
    FILES_TO_CHECK="${FILES_TO_CHECK#,}"

    # Run R verification inline
    R_OUTPUT=$(Rscript --vanilla -e "
      # Add local library path if available
      lib_path <- '${LIB_PATH}'
      if (nzchar(lib_path) && dir.exists(lib_path)) {
        .libPaths(c(lib_path, .libPaths()))
      }

      files_str <- '${FILES_TO_CHECK}'
      pairs <- strsplit(files_str, ',')[[1]]
      errors <- character(0)

      # Expected columns for Sinthetized_df
      synth_expected_cols <- c(
        'population_size', 'iteration_number', 'run_number',
        'pcrossover', 'pmutation',
        'control_optimization_horizon', 'control_implementation_horizon',
        'control_optimization_anticipation',
        'control_type', 'optimization_aim',
        'flexibility_event_length_max', 'flexibility_recover_timespan',
        'thermal_stabilization_timespan', 'minimum_flexibility',
        'flexibility_splits', 'Alpha_confort', 'month_subset',
        'Elec_total', 'Elec_Cost', 'Comfort', 'reward', 'process_time'
      )

      # Expected columns for Main_df (key columns that must exist)
      main_expected_cols <- c(
        'Elec_total', 'Elec_Cost', 'Comfort', 'Reward'
      )

      for (p in pairs) {
        parts <- strsplit(p, ':')[[1]]
        ftype <- parts[1]
        fpath <- parts[2]

        df <- tryCatch({
          if (grepl('\\\\.rds$', fpath)) {
            readRDS(fpath)
          } else {
            read.csv(fpath, stringsAsFactors = FALSE)
          }
        }, error = function(e) {
          errors <<- c(errors, paste0('  Read error (', basename(fpath), '): ', e\$message))
          return(NULL)
        })

        if (is.null(df)) next

        # Check it is a data.frame
        if (!is.data.frame(df)) {
          errors <- c(errors, paste0('  Not a data.frame: ', basename(fpath)))
          next
        }

        # Check row count > 0
        if (nrow(df) == 0) {
          errors <- c(errors, paste0('  Empty dataframe: ', basename(fpath)))
          next
        }

        # Check expected columns based on file type
        if (grepl('^synth', ftype)) {
          missing_cols <- setdiff(synth_expected_cols, names(df))
          if (length(missing_cols) > 0) {
            errors <- c(errors, paste0('  Missing columns in ', basename(fpath),
                                       ': ', paste(missing_cols, collapse=', ')))
          }

          # Check numeric types for key result columns
          numeric_cols <- c('Elec_total', 'Elec_Cost', 'Comfort',
                            'reward', 'process_time')
          for (col in intersect(numeric_cols, names(df))) {
            if (!is.numeric(df[[col]])) {
              errors <- c(errors, paste0('  Non-numeric column ', col,
                                         ' in ', basename(fpath)))
            }
          }
        } else {
          missing_cols <- setdiff(main_expected_cols, names(df))
          if (length(missing_cols) > 0) {
            errors <- c(errors, paste0('  Missing columns in ', basename(fpath),
                                       ': ', paste(missing_cols, collapse=', ')))
          }
        }

        # Check for NaN values in all numeric columns
        num_cols <- names(df)[sapply(df, is.numeric)]
        for (col in num_cols) {
          n_nan <- sum(is.nan(df[[col]]))
          n_na  <- sum(is.na(df[[col]]))
          if (n_nan > 0) {
            errors <- c(errors, paste0('  NaN values in column ', col,
                                       ' of ', basename(fpath), ' (', n_nan, ')'))
          }
          if (n_na > n_nan) {
            errors <- c(errors, paste0('  NA values in column ', col,
                                       ' of ', basename(fpath), ' (', n_na - n_nan, ')'))
          }
        }
      }

      if (length(errors) > 0) {
        cat('FAIL\n')
        cat(paste(errors, collapse='\n'), '\n')
      } else {
        cat('PASS\n')
      }
    " 2>&1)

    # Parse R verification result
    if echo "$R_OUTPUT" | head -n 1 | grep -q "^PASS"; then
      : # All good
    else
      CONFIG_OK=0
      # Capture all lines after FAIL as error messages
      R_ERRORS=$(echo "$R_OUTPUT" | tail -n +2)
      if [[ -n "$R_ERRORS" ]]; then
        ERROR_MSGS="${ERROR_MSGS}${R_ERRORS}\n"
      else
        ERROR_MSGS="${ERROR_MSGS}  Content verification failed (R error)\n"
      fi
    fi
  fi

  # ------------------------------------------------------------------
  # 6.3 Report result for this configuration
  # ------------------------------------------------------------------
  if [[ "$CONFIG_OK" -eq 1 ]]; then
    PASS_COUNT=$(( PASS_COUNT + 1 ))
  else
    FAIL_COUNT=$(( FAIL_COUNT + 1 ))
    echo "$LINE" >> "$FAILED_FILE"
    echo "FAIL [config $CONT_001] suffix=$SUFFIX"
    echo -e "$ERROR_MSGS"
  fi

  # Progress indicator every 50 configurations
  if (( CONT_001 % 50 == 0 )); then
    echo "Progress: $CONT_001 / $N verified (PASS=$PASS_COUNT, FAIL=$FAIL_COUNT)"
  fi

done < <(tail -n +2 "$INPUT_FILE")

# --- 7. Update Optim_parameters.csv with only failed configs --------
if [[ "$FAIL_COUNT" -gt 0 ]]; then
  cp "$FAILED_FILE" "$INPUT_FILE"
  echo "==================================="
  echo "Updated $INPUT_FILE with $FAIL_COUNT failed configurations."
  echo "Backup of original: $BACKUP_FILE"
else
  echo "==================================="
  echo "All $N configurations passed verification."
  echo "Optim_parameters.csv remains unchanged."
fi

rm -f "$FAILED_FILE"

# --- 8. Summary report ----------------------------------------------
echo "==================================="
echo "Verification completed at $(date)"
echo "Total configurations: $N"
echo "PASSED: $PASS_COUNT"
echo "FAILED: $FAIL_COUNT"
echo "==================================="
