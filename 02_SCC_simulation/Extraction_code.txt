#!/bin/bash
# -------------------------------------------------------------
# Console script: Extraction_code.txt
# Part of the Model Predictive Control in buildings repository
# https://github.com/robgaray/Model-Predictive-Control-in-Buildings_WORK4
# Developed by Roberto Garay Martinez
# -------------------------------------------------------------
# Location: 02_SCC_simulation/Extraction_code.txt
# -------------------------------------------------------------
# Overall description
# This script copies simulation results and logs from the fast
# scratch filesystem (/scratch/<USERNAME>/<JOB_NAME>) back to
# the persistent DIPC home directory
# (/dipc/<USERNAME>/<JOB_NAME>).
#
# It works regardless of the user's current directory by
# resolving paths from scc_settings.sh.
# -------------------------------------------------------------
# Actions performed:
#   1. Resolve the script's own directory to locate
#      scc_settings.sh
#   2. Load SCC settings (JOB_NAME, SCC_USERNAME)
#   3. Build source (/scratch) and destination (/dipc) paths
#   4. Copy logs from /scratch to /dipc
#   5. Copy simulation output files from /scratch to /dipc
# -------------------------------------------------------------
# Usage:
#   bash "02_SCC_simulation/Extraction_code.txt"
#   # or from any directory:
#   bash "/scratch/<USERNAME>/<JOB_NAME>/02_SCC_simulation/Extraction_code.txt"
# -------------------------------------------------------------
# Inputs
#   02_SCC_simulation/scc_settings.sh (relative to project root)
# -------------------------------------------------------------
# Outputs
#   Files copied from /scratch to /dipc:
#     - logs/*
#     - 01_Simulation/90_Output/*
#     - 02_SCC_simulation/job_tracker.csv
# -------------------------------------------------------------

# --- 1. Resolve script location to find project root --------------
# This ensures the script works regardless of the user's current
# working directory.
SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
PROJECT_ROOT="$(cd "$SCRIPT_DIR/.." && pwd)"

# --- Convert DOS (CRLF) to Unix (LF) line endings ---------------
sed -i 's/\r$//' "${SCRIPT_DIR}/Extraction_code.txt" 2>/dev/null
[[ -f "${SCRIPT_DIR}/scc_settings.sh" ]] && sed -i 's/\r$//' "${SCRIPT_DIR}/scc_settings.sh"

# --- 2. Load SCC settings -----------------------------------------
SCC_SETTINGS="${SCRIPT_DIR}/scc_settings.sh"
if [[ -f "$SCC_SETTINGS" ]]; then
  source "$SCC_SETTINGS"
else
  echo "Warning: $SCC_SETTINGS not found. Using defaults."
  JOB_NAME="MPC_optim"
  SCC_USERNAME=""
fi

: "${JOB_NAME:=MPC_optim}"
: "${SCC_USERNAME:=$USER}"

DIPC_USER="${SCC_USERNAME:-$USER}"

# --- 3. Build source and destination paths -------------------------
SRC_DIR="/scratch/${DIPC_USER}/${JOB_NAME}"
DST_DIR="/dipc/${DIPC_USER}/${JOB_NAME}"

echo "==================================="
echo "Extraction started at $(date)"
echo "Source (scratch): $SRC_DIR"
echo "Destination (dipc): $DST_DIR"
echo "==================================="

# Validate source directory
if [[ ! -d "$SRC_DIR" ]]; then
  echo "Error: Source directory does not exist: $SRC_DIR" >&2
  exit 1
fi

# --- 4. Copy logs ---------------------------------------------------
echo ""
echo "--- Copying logs ---"
if [[ -d "${SRC_DIR}/logs" ]]; then
  mkdir -p "${DST_DIR}/logs"
  rsync -av "${SRC_DIR}/logs/" "${DST_DIR}/logs/"
  echo "Logs copied successfully."
else
  echo "No logs directory found in $SRC_DIR"
fi

# --- 5. Copy simulation output files --------------------------------
echo ""
echo "--- Copying simulation results ---"
SRC_OUTPUT="${SRC_DIR}/01_Simulation/90_Output"
DST_OUTPUT="${DST_DIR}/01_Simulation/90_Output"
if [[ -d "$SRC_OUTPUT" ]]; then
  mkdir -p "$DST_OUTPUT"
  rsync -av "${SRC_OUTPUT}/" "${DST_OUTPUT}/"
  echo "Simulation results copied successfully."
else
  echo "No output directory found: $SRC_OUTPUT"
fi

# --- 6. Copy job tracker --------------------------------------------
echo ""
echo "--- Copying job tracker ---"
SRC_TRACKER="${SRC_DIR}/02_SCC_simulation/job_tracker.csv"
DST_SCC_DIR="${DST_DIR}/02_SCC_simulation"
if [[ -f "$SRC_TRACKER" ]]; then
  mkdir -p "$DST_SCC_DIR"
  cp -v "$SRC_TRACKER" "${DST_SCC_DIR}/job_tracker.csv"
  echo "Job tracker copied."
else
  echo "No job tracker found: $SRC_TRACKER"
fi

# --- 7. Copy verification backup (if it exists) ---------------------
echo ""
echo "--- Copying verification backups ---"
BACKUPS_FOUND=0
for bak in "${SRC_DIR}/02_SCC_simulation/Optim_parameters_backup_"*.csv; do
  if [[ -f "$bak" ]]; then
    mkdir -p "$DST_SCC_DIR"
    cp -v "$bak" "${DST_SCC_DIR}/"
    BACKUPS_FOUND=$(( BACKUPS_FOUND + 1 ))
  fi
done
if [[ "$BACKUPS_FOUND" -eq 0 ]]; then
  echo "No verification backup files found."
else
  echo "$BACKUPS_FOUND backup file(s) copied."
fi

# --- 8. Summary ------------------------------------------------------
echo ""
echo "==================================="
echo "Extraction completed at $(date)"
echo "Results are available at: $DST_DIR"
echo "==================================="
